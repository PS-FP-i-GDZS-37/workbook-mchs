    // --- Хранилище и утилиты ---
    const LS_KEY_PROFILE = 'mchs_profile';
    const LS_KEY_DIARY = 'mchs_diary';
    const LS_KEY_RESULTS = 'mchs_results';

    function saveProfile(p) { localStorage.setItem(LS_KEY_PROFILE, JSON.stringify(p)); }
    function getProfile() { const s = localStorage.getItem(LS_KEY_PROFILE); return s ? JSON.parse(s) : null; }

    function addDiary(d) {
      const list = getOrCreateList(LS_KEY_DIARY);
      list.push(d);
      localStorage.setItem(LS_KEY_DIARY, JSON.stringify(list));
    }
    function getDiary() { return getOrCreateList(LS_KEY_DIARY); }

    function addResult(r) {
      const list = getOrCreateList(LS_KEY_RESULTS);
      list.push(r);
      localStorage.setItem(LS_KEY_RESULTS, JSON.stringify(list));
    }
    function getResults() { return getOrCreateList(LS_KEY_RESULTS); }

    function getOrCreateList(key) {
      const s = localStorage.getItem(key);
      return s ? JSON.parse(s) : [];
    }

    // --- Переключение вкладок ---
    function switchTab(tabId) {
      document.querySelectorAll('section').forEach(el => el.classList.remove('active'));
      document.querySelectorAll('nav button').forEach(el => el.classList.remove('active'));
      document.getElementById(tabId).classList.add('active');
      document.getElementById('btn-' + tabId).classList.add('active');
    }

    // --- Профиль и расчёты ---
    document.getElementById('profile-form').onsubmit = function(e) {
      e.preventDefault();
      const p = {
        name: document.getElementById('name').value,
        course: +document.getElementById('course').value,
        platoon: document.getElementById('platoon').value,
        age: +document.getElementById('age').value,
        height: +document.getElementById('height').value,
        weight: +document.getElementById('weight').value
      };
      saveProfile(p);
      updateProfileView(p);
    };

    function updateProfileView(p) {
      // ИМТ
      const bmi = p.weight / ((p.height / 100) ** 2);
      document.getElementById('bmi').innerText = bmi.toFixed(1);

      // Возрастная группа по Приказу №153
      let ag;
      if (p.age <= 25) ag = '1';
      else if (p.age <= 29) ag = '2';
      else if (p.age <= 34) ag = '3';
      else if (p.age <= 39) ag = '4';
      else if (p.age <= 44) ag = '5';
      else if (p.age <= 49) ag = '6';
      else ag = '7';
      document.getElementById('age-group').innerText = ag;

      // ЧСС max и зоны по Карвонену
      const hrMax = 220 - p.age;
      const hrRest = 60; // условно
      const reserve = hrMax - hrRest;
      function range(lo, hi) { return Math.floor(hrRest + lo * reserve) + '–' + Math.floor(hrRest + hi * reserve); }
      document.getElementById('hr-max').innerText = hrMax;
      document.getElementById('zone1').innerText = range(0.5, 0.6);
      document.getElementById('zone2').innerText = range(0.6, 0.7);
      document.getElementById('zone3').innerText = range(0.7, 0.8);
      document.getElementById('zone4').innerText = range(0.8, 0.9);
      document.getElementById('zone5').innerText = range(0.9, 1.0);
    }

    // Загрузка профиля при старте
    const saved = getProfile();
    if (saved) {
      document.getElementById('name').value = saved.name;
      document.getElementById('course').value = '' + saved.course;
      document.getElementById('platoon').value = saved.platoon;
      document.getElementById('age').value = '' + saved.age;
      document.getElementById('height').value = '' + saved.height;
      document.getElementById('weight').value = '' + saved.weight;
      updateProfileView(saved);
    }
